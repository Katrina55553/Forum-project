# 第四章：数据库操作层 — crud.py

crud.py 是后端的"数据操作工具箱"——**只关心怎么读写数据库**，不关心 HTTP 请求/响应。全篇 494 行，是整个后端最厚的一个文件。

这种分层的好处：日后加 CLI 工具、后台脚本、或单元测试时，直接调用 crud 函数就行。权限校验、参数校验、HTML 过滤这些"HTTP 世界的事"全都不在这里。

## 操作数据库的三板斧

```python
db.add(obj)       # 告诉 SQLAlchemy：我要插入/更新这条记录
db.commit()       # 真正把更改写入数据库（事务提交）
db.refresh(obj)   # 从数据库重新读取（拿到自增的 id、默认值）
```

`crud.py` 里绝大多数函数都遵循同一个骨架：`add` → `commit` → `refresh` → `return`。

## 用户相关

```python
def create_user(db: Session, username: str, password: str) -> User:
    user = User(username=username, password_hash=hash_password(password))
    db.add(user)
    db.commit()
    db.refresh(user)
    return user

def get_user_by_username(db: Session, username: str) -> User | None:
    return db.query(User).filter_by(username=username).first()

def update_user(db: Session, user: User, data: dict) -> User:
    for field in ("avatar", "bio", "github_url"):
        if data.get(field) is not None:
            setattr(user, field, data[field])   # 动态设置属性
    db.commit()
    db.refresh(user)
    return user

def change_password(db: Session, user: User, new_password: str) -> None:
    user.password_hash = hash_password(new_password)
    db.commit()
```

`setattr(user, field, value)` 等价于 `user.avatar = "xxx"`，但用变量动态指定字段名。这里用白名单元组 `("avatar", "bio", "github_url")` 显式列一遍字段，**而不是遍历 `data` 里的所有 key**——防止客户端传入 `is_admin`、`password_hash` 这类越权字段。

`change_password` 单独成函数（而不是塞进 `update_user`），是为了让"改密码"这条路径只碰 `password_hash`，职责更窄。

## 标签：先查后建 + slug 规范化

```python
def get_or_create_tags(db: Session, tag_names: list[str]) -> list[Tag]:
    """获取或创建标签，返回 Tag 对象列表"""
    tags = []
    for name in tag_names:
        name = name.strip()
        if not name:
            continue
        slug = name.lower().replace(" ", "-")
        tag = db.query(Tag).filter_by(slug=slug).first()
        if not tag:
            tag = Tag(name=name, slug=slug)
            db.add(tag)
            db.flush()
        tags.append(tag)
    return tags
```

三个要点：

- **查重看 slug 而不是 name**：`slug = name.lower().replace(" ", "-")`。这样 `"Python"`、`"python"`、`"PYTHON"` 都会得到 slug `python`，被认定为同一个标签；而 `name` 保留用户第一次输入时的原始大小写
- **`db.flush()` 而不是 `db.commit()`**：只把新标签写出数据库拿到 id，但**不提交事务**。因为调用它的 `create_topic` 还要继续写帖子，两者必须在同一个事务里
- **空名跳过**：`strip()` 后为空就 `continue`，避免创建出名字为空的标签

```python
def get_all_tags(db: Session) -> list[dict]:
    """获取所有标签及其帖子数量"""
    rows = (
        db.query(Tag, func.count(topic_tags.c.topic_id).label("count"))
        .outerjoin(topic_tags, Tag.id == topic_tags.c.tag_id)
        .group_by(Tag.id)
        .order_by(func.count(topic_tags.c.topic_id).desc())
        .all()
    )
    return [{"id": t.id, "name": t.name, "slug": t.slug, "count": c} for t, c in rows]
```

`get_all_tags` 顺便统计每个标签下的帖子数：`outerjoin` 关联表 + `group_by` 标签 + `count`。用 **outerjoin 而不是 inner join**，是为了让"还没有任何帖子使用"的新标签也能出现在列表里（count 为 0）。

## 帖子相关

### 创建帖子（标签 + 原子计数）

```python
def create_topic(db: Session, author_id: int, title: str, content: str, tag_names: list[str] = None) -> Topic:
    topic = Topic(title=title, content=content, author_id=author_id)
    db.add(topic)
    if tag_names:
        tags = get_or_create_tags(db, tag_names)
        topic.tags = tags
    db.query(User).filter_by(id=author_id).update({"topic_count": User.topic_count + 1})
    db.commit()
    db.refresh(topic)
    return topic
```

一次 `create_topic` 干了三件事，但只用**一次 commit**：插入帖子、挂上标签、给作者计数 +1。中间 `get_or_create_tags` 里用的是 `flush`，所以它们天然在同一个事务中——要么全成功，要么全回滚。

**为什么不用 `user.topic_count += 1`？** 并发问题。两个请求同时读到 5，各自改成 6，丢了一次计数。用 `update({"topic_count": User.topic_count + 1})` 让数据库执行 `SET topic_count = topic_count + 1`，在数据库层面完成自增，**并发安全**。这是全篇反复出现的模式。

### LIKE 注入防护：`_escape_like`

```python
def _escape_like(s: str) -> str:
    """转义 LIKE/ILIKE 的特殊字符 % 和 _"""
    return s.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_")
```

搜索功能用 `ILIKE` 做模糊匹配。但用户输入是自由的——如果用户搜索 `100%`，`%` 会被 SQL 当成"任意字符"通配符；搜索 `a_b` 时 `_` 会匹配任意单字符。这会让搜索结果莫名其妙，甚至被用来构造昂贵的通配查询。

`_escape_like` 把 `\`、`%`、`_` 三个特殊字符转义；调用时配合 `ilike(..., escape="\\")` 告诉数据库"反斜杠是转义符"。注意顺序：**先转义反斜杠本身**，否则后面插入的 `\%` 会被二次转义。

### 帖子列表（分页 + 搜索 + 标签筛选 + 预加载）

```python
def get_topics(db: Session, page: int = 1, size: int = 10, q: str = "", tag: str = ""):
    # 先用简单查询计数（避免 joinedload 导致 count 膨胀）
    count_query = db.query(Topic)
    if q:
        escaped_q = _escape_like(q)
        count_query = count_query.filter(
            or_(
                Topic.title.ilike(f"%{escaped_q}%", escape="\\"),
                Topic.content.ilike(f"%{escaped_q}%", escape="\\"),
            )
        )
    if tag:
        count_query = count_query.filter(Topic.tags.any(Tag.slug == tag))
    total = count_query.count()

    # 再用 joinedload 查询实际数据
    query = db.query(Topic).options(joinedload(Topic.author), joinedload(Topic.tags))
    ...
    topics = (
        query.order_by(Topic.is_pinned.desc(), Topic.created_at.desc())
        .offset((page - 1) * size)
        .limit(size)
        .all()
    )
```

这个函数有一个非常重要的细节，值得单独拎出来讲：**它查了两遍**。

第一遍用干净的 `db.query(Topic)` 只做 `count()`；第二遍才带上 `joinedload(Topic.author)` 和 `joinedload(Topic.tags)` 取实际数据。为什么不合并成一次？

因为 `joinedload` 是**用 JOIN 实现的**。一篇帖子如果同时关联作者和 3 个标签，JOIN 之后会变成 3 行。`count()` 数的是 JOIN 结果的行数，于是"1 篇帖子"被数成"3 篇"——这就是所谓的 **count 膨胀**。所以计数必须用不带 JOIN 的查询。

**排序规则**：`Topic.is_pinned.desc(), Topic.created_at.desc()`。先按置顶倒序（`True` 在 `False` 前面，置顶帖永远排最前），同一档内再按发布时间倒序。这就是"置顶"功能的全部实现——数据里只存一个布尔值，排序时兼顾它。

**标签筛选**：`Topic.tags.any(Tag.slug == tag)` 而不是 JOIN 后再 filter，`any()` 会生成 `EXISTS` 子查询，语义是"这个帖子有任意一个标签的 slug 等于 X"，不会引起前面的膨胀问题。

### N+1 查询问题

不加 `joinedload`：

```
SELECT * FROM topics;              -- 1 次
SELECT * FROM users WHERE id = x;  -- N 次（每篇帖子查一次作者）
```

加了 `joinedload`，用 LEFT JOIN 一次查出所有关联数据，1 次查询搞定。

### 帖子详情（四层预加载）

```python
def get_topic_by_id(db: Session, topic_id: int) -> Topic | None:
    return (
        db.query(Topic)
        .options(
            joinedload(Topic.author),
            joinedload(Topic.comments).joinedload(Comment.author),
            joinedload(Topic.likes),
            joinedload(Topic.tags),
        )
        .filter_by(id=topic_id)
        .first()
    )
```

链式预加载：Topic → comments → Comment.author，再加作者、点赞、标签，四条关联路径一次性拉取。详情页要显示"作者的头像、每条评论的评论人、点赞数、标签"，如果让 ORM 懒加载，一个帖子 50 条评论就会产生 50+ 次查询。预加载把这一切压成一次 JOIN。

### 编辑专用的轻量查询

```python
def get_topic_for_edit(db: Session, topic_id: int) -> Topic | None:
    return db.query(Topic).options(joinedload(Topic.author)).filter_by(id=topic_id).first()
```

编辑、删除、点赞、置顶等场景只需要判断"帖子存不存在 + 作者是谁"，不需要评论和标签。所以单独准备了一个轻量查询，避免无谓的 JOIN。

### 更新帖子（支持改标签）

```python
def update_topic(db: Session, topic: Topic, data: dict) -> Topic:
    for field in ("title", "content"):
        if data.get(field) is not None:
            setattr(topic, field, data[field])
    if "tags" in data and data["tags"] is not None:
        tags = get_or_create_tags(db, data["tags"])
        topic.tags = tags
    db.commit()
    db.refresh(topic)
    return topic
```

更新时间时，标签的处理和普通字段不一样：**标签是全量替换**。`topic.tags = tags` 会把旧的关联全部清掉、换成新的。注意这里判断的是 `"tags" in data and data["tags"] is not None`——只有当客户端明确传了 `tags` 字段时才动标签，否则保持原样（区分"没传 tags"和"传了空数组表示清空标签"）。

### 删除帖子（计数兜底）

```python
def delete_topic(db: Session, topic: Topic) -> None:
    user_id = topic.author_id
    db.delete(topic)
    db.query(User).filter_by(id=user_id).update(
        {"topic_count": case((User.topic_count > 0, User.topic_count - 1), else_=0)}
    )
    db.commit()
```

删帖要把作者的 `topic_count` 减 1。但这里**没有**直接写 `User.topic_count - 1`，而是套了一层 `case`：

```python
case((User.topic_count > 0, User.topic_count - 1), else_=0)
```

意思是"如果计数大于 0 就减 1，否则保持 0"。为什么？因为冗余计数在历史数据、并发或异常删除后可能已经变成 0 甚至负数，如果无脑 `-1` 会把它减成 `-1`，之后前端显示"发了 -1 个帖子"就很难看。`case` 是一个便宜又稳妥的兜底。

同样地，`delete_comment` 里减 `comment_count` 也用了这个写法。

## 点赞（幂等设计）

```python
def like_topic(db: Session, user_id: int, topic_id: int) -> dict:
    from sqlalchemy.exc import IntegrityError
    try:
        db.execute(likes.insert().values(user_id=user_id, topic_id=topic_id))
        db.commit()
    except IntegrityError:       # 联合主键冲突 → 重复点赞
        db.rollback()            # 忽略，不报错
    count = db.query(func.count(likes.c.user_id)).filter(likes.c.topic_id == topic_id).scalar()
    return {"liked": True, "likes_count": count}

def unlike_topic(db: Session, user_id: int, topic_id: int) -> dict:
    db.execute(likes.delete().where(likes.c.user_id == user_id, likes.c.topic_id == topic_id))
    db.commit()
    count = db.query(func.count(likes.c.user_id)).filter(likes.c.topic_id == topic_id).scalar()
    return {"liked": False, "likes_count": count}
```

- `likes.c.user_id` 的 `.c` 表示 Column（因为 likes 是 Table 对象不是 ORM 类）
- **幂等性**：多次操作结果不变，快速连点两次也不报错。用户连点两次"点赞"时，第二次插入会撞上联合主键（`IntegrityError`），代码捕获后 `rollback()` 假装无事发生，最后照常返回当前点赞总数
- **`IntegrityError` 在函数内部 import**：`from sqlalchemy.exc import IntegrityError` 写在函数体内，是能省则省的小技巧
- 无论成功还是冲突，最后都**重新 COUNT 一次**返回真实点赞数，这样前端拿到的数字永远和数据库一致

## 评论 — 最复杂的操作

```python
def create_comment(db: Session, user_id: int, topic_id: int, content: str, parent_id: int | None = None) -> Comment:
    if parent_id:
        parent = db.query(Comment).filter_by(id=parent_id, topic_id=topic_id).first()
        if not parent:
            raise ValueError("Parent comment not found under this topic")
    comment = Comment(content=content, topic_id=topic_id, user_id=user_id, parent_id=parent_id)
    db.add(comment)
    db.query(User).filter_by(id=user_id).update({"comment_count": User.comment_count + 1})

    # Notify topic author (don't self-notify)
    topic = db.query(Topic).filter_by(id=topic_id).first()
    if topic and topic.author_id != user_id:
        notif = Notification(
            user_id=topic.author_id,
            type="reply",
            topic_id=topic_id,
            comment_id=comment.id,
        )
        db.add(notif)

    # Notify parent comment author (don't self-notify, don't double-notify topic author)
    if parent_id and parent and parent.user_id != user_id and parent.user_id != topic.author_id:
        notif = Notification(
            user_id=parent.user_id,
            type="reply",
            topic_id=topic_id,
            comment_id=comment.id,
        )
        db.add(notif)

    # 单事务提交，保证评论与通知的原子性
    db.commit()
    db.refresh(comment)
    return comment
```

这个函数一次干了四件事，但**只在最后 commit 一次**：

1. 校验父评论存在、且属于同一个帖子
2. 插入评论 + 作者评论数原子 +1
3. 通知帖子作者
4. 通知被回复评论的作者

**为什么一定要单事务？** 想象一下：评论写进去了、通知却因为异常没写进去。用户看到自己的评论发出去了，但被回复的人永远收不到通知——数据不一致，而且很难排查。把评论和通知放进同一个事务，要么都成功、要么都回滚，这才符合"回复某人"这个业务动作的原子语义。

`raise ValueError` 由路由层捕获后转成 400（见第六章），因为 crud 层不该知道 HTTP 状态码。

### 通知去重逻辑

| 场景 | 通知帖子作者 | 通知评论作者 |
|------|:----------:|:----------:|
| 回复自己的帖子 | ❌ | — |
| 回复别人的帖子（顶层） | ✅ | — |
| 回复别人的评论（评论作者≠帖子作者） | ✅ | ✅ |
| 回复别人的评论（评论作者=帖子作者） | ✅ | ❌ 不重复 |

三条规则对应代码里三个条件：

- `topic.author_id != user_id` — 自己回复自己不通知
- `parent.user_id != user_id` — 回复自己的评论不通知
- `parent.user_id != topic.author_id` — 如果被回复的评论作者**就是**帖子作者，那上一条已经通知过帖子作者了，不再重复发一条

### 删除评论（递归收集子回复）

```python
def delete_comment(db: Session, comment: Comment) -> None:
    # 收集所有子评论（包括嵌套的）
    def _collect_replies(c):
        result = [c]
        for r in (c.replies or []):
            result.extend(_collect_replies(r))
        return result

    db.refresh(comment)
    all_comments = _collect_replies(comment)

    # 为每个评论作者减少计数
    for c in all_comments:
        db.query(User).filter_by(id=c.user_id).update(
            {"comment_count": case((User.comment_count > 0, User.comment_count - 1), else_=0)}
        )
    db.delete(comment)
    db.commit()
```

删一条评论要连带删除它下面的**所有嵌套回复**（可能是"回复的回复的回复"）。这里用了一个内部递归函数 `_collect_replies` 把整棵子树收集成一个列表，然后：

- 逐个把每个评论作者的 `comment_count` 减 1（同样用 `case` 兜底不小于 0）
- 只需要 `db.delete(comment)` 删掉根节点——模型上 `replies` 关系配了 `cascade="all, delete-orphan"`，ORM 会自动级联删除所有子孙

`db.refresh(comment)` 是为了确保 `comment.replies` 已经被加载，否则递归时访问懒加载关系会拿到空列表，导致漏删计数。

### 构建评论树

```python
def build_comment_tree(comments: list[Comment]) -> list[dict]:
    lookup = {}
    roots = []

    # 第一遍：所有评论放入字典，key = id
    for c in comments:
        node = {
            "id": c.id,
            "content": c.content,
            "topic_id": c.topic_id,
            "user_id": c.user_id,
            "username": c.username,
            "parent_id": c.parent_id,
            "created_at": c.created_at,
            "replies": [],
        }
        lookup[c.id] = node

    # 第二遍：按 parent_id 组织父子关系
    for node in lookup.values():
        pid = node["parent_id"]
        if pid and pid in lookup:
            lookup[pid]["replies"].append(node)
        else:
            roots.append(node)

    return roots
```

O(n) 算法，用字典按 id 索引，而非嵌套循环。

```
输入扁平评论：
  [{id:1, parent_id:null}, {id:2, parent_id:1}, {id:3, parent_id:1}]

输出树形结构：
  [{id:1, replies: [{id:2}, {id:3}]}]
```

注意 `"username": c.username` 这一行——它直接读取第二章里 Comment 模型上那个 `@property`。数据库查询出来的是一张扁平表，树形结构是在内存里拼的；如果按"每层递归再查一次数据库"去实现，深度一大就会产生指数级查询，所以**一次查全、内存拼树**才是正确做法。

```python
    return db.query(func.count(Notification.id)).filter_by(user_id=user_id, is_read=False).scalar()

def mark_notification_read(db: Session, notif_id: int, user_id: int) -> None:
    db.query(Notification).filter_by(id=notif_id, user_id=user_id).update({"is_read": True})
    db.commit()

def mark_all_notifications_read(db: Session, user_id: int) -> None:
    db.query(Notification).filter_by(user_id=user_id, is_read=False).update({"is_read": True})
    db.commit()
```

- `mark_notification_read` 的 `filter_by(id=notif_id, user_id=user_id)` **同时限定归属**，防止用户把自己的通知 id 改一改就去标记别人的通知
- `mark_all_notifications_read` 用一条 UPDATE 批量处理，而不是把每条通知查出来逐条改
- 未读数查询正好命中第二章声明的 `(user_id, is_read)` 复合索引

## 私信 — 两个子查询拼会话列表

```python
def send_message(db: Session, sender_id: int, receiver_username: str, content: str) -> Message | None:
    receiver = db.query(User).filter_by(username=receiver_username).first()
    if not receiver or receiver.id == sender_id:
        return None
    message = Message(sender_id=sender_id, receiver_id=receiver.id, content=content)
    db.add(message)
    db.commit()
    db.refresh(message)
    return message
```

`send_message` 接收的是**用户名**而不是 id（对接口更友好），所以先查一次用户。注意它的两个返回 `None` 的情况：**收件人不存在**、**给自己发私信**。返回 `None` 而不是抛异常，是因为"用户不存在"和"不能给自己发"在 HTTP 层都归为基础校验失败，由路由层统一转成 400。

```python
def get_conversations(db: Session, user_id: int) -> list[dict]:
    """获取会话列表（按最后消息时间排序）"""
    # 子查询：获取每个对话用户的最新消息 ID
    latest_sq = (
        db.query(
            case(
                (Message.sender_id == user_id, Message.receiver_id),
                else_=Message.sender_id
            ).label("other_id"),
            func.max(Message.id).label("last_msg_id"),
        )
        .filter(or_(Message.sender_id == user_id, Message.receiver_id == user_id))
        .group_by("other_id")
        .subquery()
    )

    # 子查询：每个对话用户的未读消息数
    unread_sq = (
        db.query(
            Message.sender_id.label("other_id"),
            func.count(Message.id).label("cnt"),
        )
        .filter(Message.receiver_id == user_id, Message.is_read == False)
        .group_by(Message.sender_id)
        .subquery()
    )

    # 主查询：最新消息 + 用户信息 + 未读数
    rows = (
        db.query(
            User.username, User.avatar, Message.content, Message.created_at,
            func.coalesce(unread_sq.c.cnt, 0).label("unread_count"),
        )
        .join(latest_sq, User.id == latest_sq.c.other_id)
        .join(Message, Message.id == latest_sq.c.last_msg_id)
        .outerjoin(unread_sq, unread_sq.c.other_id == User.id)
        .order_by(Message.created_at.desc())
        .all()
    )
    return [
        {
            "username": r.username,
            "avatar": r.avatar or "",
            "last_message": (r.content or "")[:50],
            "last_message_at": r.created_at.isoformat() if r.created_at else "",
            "unread_count": r.unread_count or 0,
        }
        for r in rows
    ]
```

私信的难点在于：一段对话里的消息，**"我是发送方"和"我是接收方"混在一起**。会话列表要展示"对方是谁 + 最后一条消息 + 未读数"，无法用一次简单的 GROUP BY 搞定。于是拆成两个子查询：

**第一个子查询 `latest_sq`** 找出"每个对话对象的最后一条消息 id"：
- `case((Message.sender_id == user_id, Message.receiver_id), else_=Message.sender_id)` 这张"人格分裂"的表达式，把所有消息统一成"对方 id"——我发的消息取 receiver，别人发的取 sender
- 按 `other_id` 分组后取 `func.max(Message.id)`。用最大 id 而不是最大时间戳当"最新"，是因为自增 id 严格递增，不受时间精度影响

**第二个子查询 `unread_sq`** 统计每个发件人给我发的未读消息数。

**主查询**把三者拼起来：`join(latest_sq)` 拿到对话对象，`join(Message)` 拿到最后一条消息内容，`outerjoin(unread_sq)` 挂上未读数（`outerjoin` 保证没有未读的会话也在列表里，`coalesce(..., 0)` 把 NULL 变成 0）。最后按最后消息时间倒序——最近的对话排最前。

### 聊天记录与已读

```python
def get_messages_with_user(db: Session, user_id: int, other_username: str, page: int = 1, size: int = 50) -> dict | None:
    other_user = db.query(User).filter_by(username=other_username).first()
    if not other_user:
        return None
    query = db.query(Message).filter(
        or_(
            (Message.sender_id == user_id) & (Message.receiver_id == other_user.id),
            (Message.sender_id == other_user.id) & (Message.receiver_id == user_id),
        )
    )
    total = query.count()
    messages = query.order_by(Message.created_at.asc()).offset((page - 1) * size).limit(size).all()
    return {
        "messages": [ { "id": msg.id, ... } for msg in messages ],
        "other_user": {"id": other_user.id, "username": other_user.username, "avatar": other_user.avatar or ""},
        "total": total,
    }
```

查两人之间的消息用 `or_` 组合两个方向的条件。**升序**排列（最早的在前），符合聊天记录从上往下的阅读习惯，分页时 `size` 默认 50。

```python
def mark_messages_read(db: Session, user_id: int, other_username: str) -> None:
```

"标记已读"只标记**对方发给我**的消息（`sender_id == other_user.id, receiver_id == user_id`），不会误标我发给对方的消息。

## crud.py 设计模式

| 模式 | 例子 | 好处 |
|------|------|------|
| 接收 Session 参数 | `def xxx(db: Session, ...)` | 依赖注入，方便测试和复用 |
| 返回 ORM 对象或 dict | `return user` / `return {...}` | 调用方决定怎么用 |
| 不管权限校验 | 不判断"能不能操作" | 职责单一，权限集中在 main.py |
| 原子更新 | `User.count + 1` | 并发安全 |
| 计数兜底 | `case((count > 0, count - 1), else_=0)` | 防止计数被减成负数 |
| 多步骤单事务 | 评论 + 通知一次 commit | 保证数据一致性 |
| 一次查全、内存拼装 | `build_comment_tree` / `get_topics` | 避免 N+1 查询 |

## 本章要点

- crud 层只和数据库打交道，权限/参数/HTML 过滤都留给路由层
- 计数增减一律用 SQL 表达式而非读改写；减法加 `case` 兜底
- LIKE 搜索必须转义 `%`、`_`，否则用户输入会变成通配符
- `joinedload` 解决 N+1，但计数必须用不带 JOIN 的查询以免膨胀
- 幂等（点赞）靠数据库唯一约束 + 捕获 `IntegrityError`
- 评论与通知必须单事务提交，保证"评论成功就有通知"
- 复杂聚合（会话列表）用子查询在数据库侧算好，再在 Python 侧拼字典
