# 第四章：数据库操作层 — crud.py

crud.py 是后端的"数据操作工具箱"——**只关心怎么读写数据库**，不关心 HTTP 请求/响应。

这种分层的好处：日后加 CLI 工具、后台脚本、或单元测试时，直接调用 crud 函数就行。

## 操作数据库的三板斧

```python
db.add(obj)       # 告诉 SQLAlchemy：我要插入/更新这条记录
db.commit()       # 真正把更改写入数据库（事务提交）
db.refresh(obj)   # 从数据库重新读取（拿到自增的 id、默认值）
```

## 用户相关

```python
def create_user(db: Session, username: str, password: str) -> User:
    user = User(username=username, password_hash=hash_password(password))
    db.add(user)
    db.commit()
    db.refresh(user)
    return user

def get_user_by_username(db: Session, username: str) -> User | None:
    return db.query(User).filter_by(username=username).first()

def update_user(db: Session, user: User, data: dict) -> User:
    for field in ("avatar", "bio", "github_url"):
        if data.get(field) is not None:
            setattr(user, field, data[field])   # 动态设置属性
    db.commit()
    db.refresh(user)
    return user
```

`setattr(user, field, value)` 等价于 `user.avatar = "xxx"`，但用变量动态指定字段名。

## 帖子相关

### 创建帖子（原子计数）

```python
def create_topic(db, author_id, title, content):
    topic = Topic(title=title, content=content, author_id=author_id)
    db.add(topic)
    # 原子操作：直接在数据库执行 SET topic_count = topic_count + 1
    db.query(User).filter_by(id=author_id).update({"topic_count": User.topic_count + 1})
    db.commit()
    db.refresh(topic)
    return topic
```

**为什么不用 `user.topic_count += 1`？** 并发问题。两个请求同时读到 5，各自改成 6，丢了一次计数。用 `update()` 在数据库层面加锁执行，**并发安全**。

### 帖子列表（分页 + 搜索 + 预加载）

```python
def get_topics(db, page=1, size=10, q=""):
    query = db.query(Topic).options(joinedload(Topic.author))
    if q:
        query = query.filter(
            or_(Topic.title.ilike(f"%{q}%"), Topic.content.ilike(f"%{q}%"))
        )
    total = query.count()
    topics = (
        query.order_by(Topic.created_at.desc())
        .offset((page - 1) * size)
        .limit(size)
        .all()
    )
    # 每个帖子附上统计信息
    result = []
    for t in topics:
        result.append({
            "id": t.id, "title": t.title,
            "comment_count": db.query(func.count(Comment.id)).filter_by(topic_id=t.id).scalar(),
            "likes_count": db.query(func.count(likes.c.user_id)).filter(likes.c.topic_id == t.id).scalar(),
            ...
        })
    return result, total
```

### N+1 查询问题

不加 `joinedload`：

```
SELECT * FROM topics;              -- 1 次
SELECT * FROM users WHERE id = x;  -- N 次（每篇帖子查一次作者）
```

加了 `joinedload`，用 LEFT JOIN 一次查出所有关联数据，1 次查询搞定。

### 帖子详情（三层预加载）

```python
def get_topic_by_id(db, topic_id):
    return (
        db.query(Topic)
        .options(
            joinedload(Topic.author),                             # 作者
            joinedload(Topic.comments).joinedload(Comment.author), # 评论 → 评论作者
            joinedload(Topic.likes),                              # 点赞关系
        )
        .filter_by(id=topic_id)
        .first()
    )
```

链式预加载：Topic → comments → Comment.author，三层一次性拉取。

## 点赞（幂等设计）

```python
def like_topic(db, user_id, topic_id):
    try:
        db.execute(likes.insert().values(user_id=user_id, topic_id=topic_id))
        db.commit()
    except IntegrityError:       # 联合主键冲突 → 重复点赞
        db.rollback()            # 忽略，不报错
    count = db.query(...).scalar()
    return {"liked": True, "likes_count": count}

def unlike_topic(db, user_id, topic_id):
    db.execute(likes.delete().where(
        likes.c.user_id == user_id, likes.c.topic_id == topic_id
    ))
    db.commit()
    ...
```

- `likes.c.user_id` 的 `.c` 表示 Column（因为 likes 是 Table 对象不是 ORM 类）
- **幂等性**：多次操作结果不变，快速连点两次也不报错

## 评论 — 最复杂的操作

```python
def create_comment(db, user_id, topic_id, content, parent_id=None):
    # 1. 校验：如果回复某条评论，确认它存在且属于同一帖子
    if parent_id:
        parent = db.query(Comment).filter_by(id=parent_id, topic_id=topic_id).first()
        if not parent:
            raise ValueError("Parent comment not found under this topic")

    # 2. 插入评论 + 原子更新用户评论数
    comment = Comment(...)
    db.add(comment)
    db.query(User).filter_by(id=user_id).update({"comment_count": User.comment_count + 1})
    db.commit()
    db.refresh(comment)

    # 3. 通知帖子作者（自己回复自己不通知）
    topic = db.query(Topic).filter_by(id=topic_id).first()
    if topic and topic.author_id != user_id:
        db.add(Notification(user_id=topic.author_id, type="reply", ...))

    # 4. 通知被回复的评论作者（不去重通知帖子作者）
    if parent_id and parent.user_id != user_id and parent.user_id != topic.author_id:
        db.add(Notification(user_id=parent.user_id, type="reply", ...))

    db.commit()
    return comment
```

### 通知去重逻辑

| 场景 | 通知帖子作者 | 通知评论作者 |
|------|:----------:|:----------:|
| 回复自己的帖子 | ❌ | — |
| 回复别人的帖子（顶层） | ✅ | — |
| 回复别人的评论（评论作者≠帖子作者） | ✅ | ✅ |
| 回复别人的评论（评论作者=帖子作者） | ✅ | ❌ 不重复 |

## 构建评论树

```python
def build_comment_tree(comments: list[Comment]) -> list[dict]:
    lookup = {}
    roots = []

    # 第一遍：所有评论放入字典，key = id
    for c in comments:
        node = {"id": c.id, ..., "replies": []}
        lookup[c.id] = node

    # 第二遍：按 parent_id 组织父子关系
    for node in lookup.values():
        pid = node["parent_id"]
        if pid and pid in lookup:                 # 有父节点且存在
            lookup[pid]["replies"].append(node)    # 挂到父节点
        else:
            roots.append(node)                     # 顶层评论

    return roots
```

O(n) 算法，用字典按 id 索引，而非嵌套循环。

```
输入扁平评论：
  [{id:1, parent_id:null}, {id:2, parent_id:1}, {id:3, parent_id:1}]

输出树形结构：
  [{id:1, replies: [{id:2}, {id:3}]}]
```

## crud.py 设计模式

| 模式 | 例子 | 好处 |
|------|------|------|
| 接收 Session 参数 | `def xxx(db: Session, ...)` | 依赖注入，方便测试和复用 |
| 返回 ORM 对象或 dict | `return user` | 调用方决定怎么用 |
| 不管权限校验 | 不判断"能不能操作" | 职责单一，权限集中在 main.py |
| 原子更新 | `User.count + 1` | 并发安全 |
