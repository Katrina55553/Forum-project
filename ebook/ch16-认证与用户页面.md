### 表单设计

```html

  <div class="field">
    <label>用户名</label>
    <input v-model="username" type="text" autocomplete="username" required />
  </div>
  <div class="field">
    <label>密码</label>
    <input v-model="password" type="password" autocomplete="current-password" required />
  </div>
  <button type="submit" :disabled="submitting">
    {{ submitting ? "登录中..." : "登录" }}
  </button>
  <p class="hint">
    还没有账号？<router-link to="/register">注册</router-link>
  </p>
</form>
```

### 登录逻辑

```javascript
import { useAuthStore } from "../stores/auth";
import { useRoute, useRouter } from "vue-router";

const auth = useAuthStore();
const route = useRoute();
const router = useRouter();

const username = ref("");
const password = ref("");
const error = ref("");
const submitting = ref(false);

async function handleLogin() {
  error.value = "";
  submitting.value = true;
  try {
    await auth.login(username.value, password.value);
    // 登录成功后跳转到之前想去的页面，没有则去首页
    const redirect = route.query.redirect || "/";
    router.push(redirect);
  } catch {
    error.value = "用户名或密码错误";
  } finally {
    submitting.value = false;
  }
}
```

**关键点：** 登录成功后用 `route.query.redirect` 跳回之前被拦截的页面。这个参数是路由守卫在检测到未登录时传过来的。

## 注册 — RegisterView.vue

### 前端校验

```javascript
function validate() {
  if (!username.value || !password.value || !confirm.value) {
    error.value = "请填写所有字段";
    return false;
  }
  if (password.value !== confirm.value) {
    error.value = "两次密码不一致";
    return false;
  }
  return true;
}

async function handleRegister() {
  if (!validate()) return;
  submitting.value = true;
  try {
    await auth.register(username.value, password.value);
    showToast.success("注册成功，请登录");
    router.push("/login");
  } catch (err) {
    if (err.response?.status === 409) {
      error.value = "用户名已存在";
    } else {
      error.value = "注册失败，请稍后重试";
    }
  } finally {
    submitting.value = false;
  }
}
```

- 前端校验做基本检查（非空、密码一致）
- 后端校验结果通过 HTTP 状态码传达（409 冲突）

## 用户主页 — UserProfile.vue

```javascript
import { getUserProfile } from "../api/user";

const user = ref(null);
const topics = ref([]);
const loading = ref(true);
const error = ref("");

onMounted(async () => {
  try {
    const res = await getUserProfile(route.params.username);
    user.value = res.data;
    topics.value = res.data.topics;
  } catch {
    error.value = "用户不存在";
  } finally {
    loading.value = false;
  }
});
```

import { updateMe, changePassword } from "../api/auth";

async function saveProfile() {
  submitting.value = true;
  try {
    const res = await updateMe({ avatar: avatar.value, bio: bio.value, github_url: githubUrl.value });
    auth.user = res.data;                           // 更新 Pinia 状态
    localStorage.setItem("user", JSON.stringify(res.data));  // 同步到 localStorage
    showToast.success("资料已更新");
  } finally {
    submitting.value = false;
  }
}

### 密码修改

```javascript
async function changePwd() {
  if (newPwd.value !== confirmPwd.value) {
    error.value = "两次密码不一致";
    return;
  }
  try {
    await changePassword(oldPwd.value, newPwd.value);
    showToast.success("密码已修改");
    oldPwd.value = newPwd.value = confirmPwd.value = "";
  } catch (err) {
    error.value = err.response?.data?.detail || "修改失败";
  }
}
```
